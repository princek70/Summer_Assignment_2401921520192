/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class Solution {
    public boolean pathSum(TreeNode root, int targetSum,int sum,ArrayList<TreeNode> path) 
    {
        if(root==null)
        return false;
        path.add(root);
        sum+=root.val;
        if(sum==targetSum && root.left == null && root.right == null)
        return true;
        if(pathSum(root.left,targetSum,sum,path) || pathSum(root.right,targetSum,sum,path))
         return true;
        path.remove(path.size()-1);
        return false;
    }
    public boolean hasPathSum(TreeNode root, int targetSum) {
        ArrayList<TreeNode> path=new ArrayList<>();
        return pathSum(root, targetSum,0,path);
       
    }
}
