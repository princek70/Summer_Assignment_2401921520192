        if(s.length()==0 || s.length()==1)
         return s.length();
        HashSet<Character> hs=new HashSet<>();
        int max=Integer.MIN_VALUE;
        int i=0,j=0;
        while(j<s.length())
        {
            if(!hs.contains(s.charAt(j)))
            {
                hs.add(s.charAt(j));
                max=Math.max(max,hs.size());
                j++;
            }
            else
            {
               hs.remove(s.charAt(i));
               i++;
            }

        }
        return max;

    }
