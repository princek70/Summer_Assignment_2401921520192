class Solution {
    public boolean isValid(String s) {
        Stack<Character> s=new Stack<>();
        for(int i=0;i<s.length();i++)
        {
            char ch=s.charAt(i);
            if(ch=='(' || ch=='{' ||  ch=='[')
            s.push(ch);
            else
            {
                if(s.isEmpty())
                return false;
                else if(s.peek()=='(' && ch==')' ||
                        s.peek()=='{' && ch=='}'||
                        s.peek()=='[' && ch==']')
                        s.pop();
                else 
                return false;
            }
        }
        return s.isEmpty();
    }
}
