class Solution {
    public int maxSubArray(int[] nums) {
        int cs=Integer.MIN_VALUE;
        for(int i=0;i<nums.length;i++)
        {
            if(nums[i]<0)
            cs=Math.max(cs,nums[i]);
            else 
            break;
            if(i==nums.length-1)
            return cs;
        }
        cs=0;
        int ms=Integer.MIN_VALUE;
        for(int i=0;i<nums.length;i++)
        {
             cs=cs+nums[i];
            if(cs<0)
            cs=0;
            ms=Math.max(ms,cs);
        }
        return ms;
    }
}
