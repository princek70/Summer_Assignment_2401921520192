class Solution {
    public boolean containsDuplicate(int[] nums) {
        if(nums.length==1)
        return false;
        if(nums[0]==nums[1])
        return true;
        Arrays.sort(nums);
        int lp=1;
        while(lp<nums.length-1)
        {
            if(nums[lp-1]==nums[lp] || nums[lp]==nums[lp+1])
            return true;
            lp++;
        }
        return false;
        
    }
}
